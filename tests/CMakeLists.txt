cmake_minimum_required(VERSION 3.16)
project(tourbox_bridge_tests C)
enable_testing()
add_executable(test_bridge_core test_bridge_core.c ../main/bridge_core.c ../main/tourbox_advertisement.c ../main/lite_profile.c)
target_compile_definitions(test_bridge_core PRIVATE TB_TEST_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/fixtures")
target_include_directories(test_bridge_core PRIVATE ../main/include)
target_compile_options(test_bridge_core PRIVATE -Wall -Wextra -Werror)
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(test_bridge_core PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(test_bridge_core PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME bridge_core COMMAND test_bridge_core)
